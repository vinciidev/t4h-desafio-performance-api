WITH total AS (SELECT produto_id, count(*) AS itens, sum(quantidade) AS unidades, sum(quantidade * preco_unitario_centavos) AS receita FROM itens_pedido GROUP BY produto_id), cancelados AS (SELECT i.produto_id, count(*) AS itens, sum(i.quantidade) AS unidades, sum(i.quantidade * i.preco_unitario_centavos) AS receita FROM pedidos p JOIN itens_pedido i ON i.pedido_id = p.id WHERE p.status = 'cancelado' GROUP BY i.produto_id) SELECT pr.id, pr.nome, pr.categoria, t.unidades - coalesce(c.unidades, 0) AS unidades, t.receita - coalesce(c.receita, 0) AS receita_centavos FROM total t JOIN produtos pr ON pr.id = t.produto_id LEFT JOIN cancelados c ON c.produto_id = t.produto_id WHERE t.itens > coalesce(c.itens, 0) ORDER BY receita_centavos DESC, pr.id LIMIT 10
